import type { AppConfig } from "../types"; export const EMPTY_CONFIG: AppConfig = { tenantId: "", clientId: "", workbookShareUrl: "", workbookDriveId: "", workbookItemId: "", tableName: "TablaControlDocumental", worksheetName: "Control documental", }; const STORAGE_KEY = "control-recoleccion-config-v1"; export async function loadConfig(): Promise { let publishedConfig: Partial = {}; try { const response = await fetch(`${import.meta.env.BASE_URL}app-config.json`, { cache: "no-store" }); if (response.ok) publishedConfig = await response.json(); } catch { // La configuración guardada en este dispositivo sigue siendo válida sin conexión. } let localConfig: Partial = {}; try { localConfig = JSON.parse(localStorage.getItem(STORAGE_KEY) || "{}"); } catch { localStorage.removeItem(STORAGE_KEY); } return { ...EMPTY_CONFIG, ...publishedConfig, ...localConfig }; } export function saveConfig(config: AppConfig) { localStorage.setItem(STORAGE_KEY, JSON.stringify(config)); } export function clearLocalConfig() { localStorage.removeItem(STORAGE_KEY); } export function isConfigured(config: AppConfig) { const hasWorkbook = Boolean(config.workbookShareUrl || (config.workbookDriveId && config.workbookItemId)); return Boolean(config.clientId && config.tenantId && hasWorkbook && config.tableName && config.worksheetName); }
